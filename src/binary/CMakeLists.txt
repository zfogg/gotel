cmake_minimum_required(VERSION 3.24)
project(gotel_binary LANGUAGES C)
add_executable(gotel main.c engine.c)
target_compile_features(gotel PRIVATE c_std_99)
target_compile_definitions(gotel PRIVATE GOTEL_SQL_STANDALONE=1 UTF8PROC_STATIC)
target_include_directories(gotel PRIVATE ../../c/include ../../go/sqltrace)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME binary COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_binary.py $<TARGET_FILE:gotel>)
endif()
install(TARGETS gotel RUNTIME DESTINATION bin)
