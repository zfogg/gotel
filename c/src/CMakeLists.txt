if(GOTEL_WITH_SQLITE)
  if(GOTEL_SQLITE_TARGET)
    if(NOT TARGET "${GOTEL_SQLITE_TARGET}")
      message(FATAL_ERROR "GOTEL_SQLITE_TARGET must name an existing CMake target")
    endif()
    set(_gotel_sqlite_target "${GOTEL_SQLITE_TARGET}")
  else()
    find_package(SQLite3 REQUIRED)
    if(TARGET SQLite3::SQLite3)
      set(_gotel_sqlite_target SQLite3::SQLite3)
    elseif(TARGET SQLite::SQLite3)
      set(_gotel_sqlite_target SQLite::SQLite3)
    else()
      message(FATAL_ERROR "FindSQLite3 did not provide a SQLite3 CMake target")
    endif()
  endif()
  target_sources(gotel PRIVATE sqlite.c named.c)
  target_link_libraries(gotel PUBLIC ${_gotel_sqlite_target})
  target_compile_definitions(gotel PUBLIC GOTEL_WITH_SQLITE=1)
endif()

option(GOTEL_BUILD_CXX "Build the C++27 facade" ON)
if(GOTEL_BUILD_CXX)
  # CMake does not yet map C++27 for Clang, even when Clang supports it.
  if(GOTEL_CXX_STANDARD STREQUAL "27" AND CMAKE_CXX_COMPILER_ID STREQUAL "Clang" AND
     NOT DEFINED CMAKE_CXX27_STANDARD_COMPILE_OPTION)
    if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
      set(_gotel_cxx27_flag "/clang:-std=c++2d")
    else()
      set(_gotel_cxx27_flag "-std=c++2d")
    endif()
    include(CheckCXXCompilerFlag)
    check_cxx_compiler_flag("${_gotel_cxx27_flag}" _gotel_has_cxx27)
    if(_gotel_has_cxx27)
      set(CMAKE_CXX27_STANDARD_COMPILE_OPTION "${_gotel_cxx27_flag}")
      set(CMAKE_CXX27_EXTENSION_COMPILE_OPTION "${_gotel_cxx27_flag}")
    endif()
  endif()
  add_library(gotel_cxx cxx.cpp)
  add_library(gotel::cxx ALIAS gotel_cxx)
  set_target_properties(gotel_cxx PROPERTIES CXX_STANDARD ${GOTEL_CXX_STANDARD} CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO POSITION_INDEPENDENT_CODE ON)
  target_include_directories(gotel_cxx PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/cxx/include>
    $<INSTALL_INTERFACE:include>)
  target_link_libraries(gotel_cxx PUBLIC gotel::gotel)
  if(_gotel_has_cxx27)
    target_compile_options(gotel_cxx PUBLIC "$<$<COMPILE_LANGUAGE:CXX>:${_gotel_cxx27_flag}>")
  endif()
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang" AND NOT MSVC)
    find_library(GOTEL_LIBCXX NAMES c++ libc++ libcxx)
    if(GOTEL_LIBCXX)
      target_compile_options(gotel_cxx PUBLIC -stdlib=libc++)
      target_link_options(gotel_cxx INTERFACE -stdlib=libc++)
    endif()
  endif()
  install(TARGETS gotel_cxx EXPORT gotelTargets)
  install(DIRECTORY cxx/include/gotel DESTINATION include)
endif()
